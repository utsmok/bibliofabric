    Client for the OpenAIRE Graph API (V4) and the Scholexplorer API.

    Entity resources are available as typed attributes; single-entity lookups go through the V3 by-id routes because V4 has none.     class AireloomClient(BaseApiClient):
    """Client for the OpenAIRE Graph API (V4) and the Scholexplorer API.

    Entity resources are available as typed attributes; single-entity lookups
    go through the V3 by-id routes because V4 has none. Authentication
    resolves in the constructor: set ``openaire_refresh_token`` /
    ``refresh_token`` for OAuth2 refresh-token access, or set both
    ``openaire_client_id`` and ``openaire_client_secret`` (settings,
    ``AIRELOOM_OPENAIRE_CLIENT_ID`` / ``AIRELOOM_OPENAIRE_CLIENT_SECRET``, or
    ``client_id`` / ``client_secret``) for OAuth2 client-credentials access
    against the Keycloak token endpoint; partial client credentials raise
    :class:`ConfigurationError`, and without credentials the client is
    anonymous. An explicit ``auth_strategy`` always wins.

    The Scholix resource talks to the Scholexplorer host through a second
    internal client that shares settings and credentials but throttles itself
    to one request per second. See the package README and docs for usage
    examples of both the async and the sync variants.
    """

    def __init__(
        self,
        settings: AireloomSettings | None = None,
        auth: AuthStrategy | None = None,
        *,
        auth_strategy: AuthStrategy | None = None,
        base_url: str | None = None,
        scholix_base_url: str | None = None,
        mailto: str | None = None,
        client_id: str | None = None,
        client_secret: str | None = None,
        refresh_token: str | None = None,
        http_client: httpx.AsyncClient | None = None,
    ) -> None:
        """Build the client from settings, resolving credentials immediately.

        ``settings`` defaults to :func:`aireloom.config.get_settings`, which
        reads ``AIRELOOM_*`` environment variables and ``.env``. Keyword
        overrides (``base_url``, ``scholix_base_url``, ``mailto``,
        ``client_id``, ``client_secret``, ``refresh_token``) take precedence
        over ``settings``. Pass ``auth_strategy`` for a custom auth strategy
        (``auth`` is a deprecated alias).
        """
        if auth is not None and auth_strategy is not None:
            raise ConfigurationError(
                "Pass either 'auth_strategy' or the deprecated 'auth' parameter, "
                "not both."
            )
        if auth is not None:
            warnings.warn(
                "The 'auth' parameter is deprecated; use 'auth_strategy' instead.",
                DeprecationWarning,
                stacklevel=2,
            )
            explicit_auth: AuthStrategy | None = auth
        else:
            explicit_auth = auth_strategy

        resolved = settings or get_settings()
        overrides: dict[str, str] = {}
        if base_url is not None:
            overrides["base_url"] = base_url
        if scholix_base_url is not None:
            overrides["scholix_base_url"] = scholix_base_url
        if mailto is not None:
            overrides["mailto"] = mailto
        if client_id is not None:
            overrides["openaire_client_id"] = client_id
        if client_secret is not None:
            overrides["openaire_client_secret"] = client_secret
        if refresh_token is not None:
            overrides["openaire_refresh_token"] = refresh_token
        if overrides:
            resolved = resolved.model_copy(update=overrides)
        strategy = (
            explicit_auth if explicit_auth is not None else _resolve_auth(resolved)
        )
        super().__init__(
            base_url=resolved.base_url,
            settings=resolved,
            unwrapper=OpenAireUnwrapper(),
            auth=strategy,
            http_client=http_client,
        )
        self._scholix_api = BaseApiClient(
            base_url=resolved.scholix_base_url,
            settings=resolved.model_copy(
                update={
                    "max_requests_per_second": resolved.scholix_max_requests_per_second
                }
            ),
            unwrapper=ScholixUnwrapper(),
            auth=strategy,
        )
        auth_label = (
            "anonymous" if isinstance(strategy, NoAuth) else type(strategy).__name__
        )
        logger.debug("%s initialized (%s).", type(self).__name__, auth_label)

    async def request(
        self,
        method: str,
        path: str,
        *,
        params: Mapping[str, Any] | None = None,
        json: Any | None = None,
        data: Mapping[str, Any] | None = None,
        content: str | bytes | None = None,
        headers: Mapping[str, str] | None = None,
        timeout: float | httpx.Timeout | None = None,
        retry: bool | None = None,
        follow_redirects: bool | None = None,
    ) -> httpx.Response:
        """Send a request through the core retry pipeline.

        V4 accepts ``mailto`` for polite-pool admission (V3 rejects it), so the
        configured contact address is attached to ``v4/`` requests only.
        """
        effective_params: Mapping[str, Any] | None = params
        if path.lstrip("/").startswith("v4/") and self._settings.mailto:
            if params is None:
                effective_params = {"mailto": self._settings.mailto}
            elif isinstance(params, Mapping):
                if "mailto" not in params:
                    effective_params = {"mailto": self._settings.mailto, **params}
            else:
                qp = httpx.QueryParams(params)
                if "mailto" not in qp:
                    effective_params = qp.set("mailto", self._settings.mailto)
        return await super().request(
            method,
            path,
            params=effective_params,
            json=json,
            data=data,
            content=content,
            headers=headers,
            timeout=timeout,
            retry=retry,
            follow_redirects=follow_redirects,
        )

        """Close both the Graph and the Scholix HTTP clients."""

            await super().aclose()

            await self._scholix_api.aclose()

    class ResearchProductsResource(GraphResource[ResearchProduct]):
    """V4 research products endpoint: search, iterate, count, facets, group_by.

    ``get(id)`` reads the V3 by-id route because V4 has no by-id endpoints.
    Batch lookups use OR expressions over ``id`` and ``ids.doi``.
    """

    path = "v4/research-products"
    model = ResearchProduct
    filters_model = ResearchProductsFilters
    sort_fields = frozenset(
        {
            "relevance",
            "publication_date",
            "date_of_collection",
            "citation_count",
            "influence",
            "popularity",
            "impulse",
        }
    )
    batch_size = 50

    def __init__(self, client: BaseApiClient, links: LinksResource) -> None:
        super().__init__(client)
        self._links = links

    def record_keys(self, entity: ResearchProduct, field: str) -> list[str]:
        if field == "ids.doi":
            # Deduplicated records carry several DOIs; batch_get keeps the requested one.
            return [normalize_doi(doi) for doi in entity.all_dois]
        return super().record_keys(entity, field)

    async def batch_get_by_doi(
        self,
        dois: list[str] | tuple[str, ...],
        *,
        batch_size: int | None = None,
        concurrency: int = 4,
    ) -> dict[str, ResearchProduct]:
        """Fetch research products for many DOIs in bounded, concurrent chunks.

        Keys are bare, lowercased DOIs; missing DOIs are absent from the
        result. The request goes out as ``filter=ids.doi:a|b|c``.
        """
        normalized = list(dict.fromkeys(normalize_doi(doi) for doi in dois))
        return await self.batch_get(
            normalized, field="ids.doi", batch_size=batch_size, concurrency=concurrency
        )

    async def by_orcid(self, orcid: str, *, limit: int = 100) -> list[ResearchProduct]:
        """Return research products whose authorship carries ``orcid``.

        Accepts any spelling ``bibliofabric.ids.normalize_orcid`` handles
        (bare, dashed, or an ``https://orcid.org/`` URL); the filter sends
        the hyphenated form via the V4 ``authorships.author.orcid`` field and
        paginates via cursor when ``limit`` exceeds 100.
        """
        page_size = min(limit, self.dialect.max_page_size) if limit > 0 else None
        return await self.collect(
            ResearchProductsFilters(authorships_author_orcid=normalize_orcid(orcid)),
            limit=limit,
            page_size=page_size,
        )

    async def get_fields(
        self,
        ids: Sequence[str],
        fields: Sequence[str],
        *,
        batch_size: int | None = None,
        concurrency: int = 4,
    ) -> list[ResearchProduct]:
        """Sparse-fetch ``fields`` for known ids in bounded, concurrent chunks.

        Combines the V4 ``select`` parameter with an ``id`` OR-filter (at most
        50 ids per chunk). The returned records carry only the requested
        fields; every other model field stays at its default.
        """
        selected = [field.strip() for field in fields if field and field.strip()]
        if not selected:
            raise ConfigurationError("get_fields() requires at least one field name.")
        size = self.batch_size if batch_size is None else batch_size
        if size <= 0 or size > MAX_OR_VALUES:
            raise ConfigurationError(
                f"batch_size must be between 1 and {MAX_OR_VALUES}, got {size}."
            )
        if concurrency <= 0:
            raise ConfigurationError(
                f"concurrency must be positive, got {concurrency}."
            )
        wanted = list(
            dict.fromkeys(norm for raw in ids if (norm := self.normalize_id(raw)))
        )
        if not wanted:
            return []
        chunks = [wanted[start : start + size] for start in range(0, len(wanted), size)]

        async def fetch_chunk(chunk: list[str]) -> list[ResearchProduct]:
            page = await self.search(
                {"id": chunk}, select=selected, page_size=len(chunk)
            )
            return page.results

        chunk_results = await run_bounded(fetch_chunk, chunks, concurrency)
        return [record for group in chunk_results for record in group]

    async def search_links(
        self,
        filters: LinksFilters | Mapping[str, Any] | None = None,
        *,
        page: int | None = None,
        page_size: int | None = None,
    ) -> Page[Relation]:
        """Fetch one page of relations from the Graph links route."""
        return await self._links.search(filters, page=page, page_size=page_size)

    def iterate_links(
        self,

    """Scholix link search over the Scholexplorer host (V3 default, V2 fallback).

    Backed by a dedicated internal API client for ``api.scholexplorer.openaire.eu``
    that shares the Graph client's credentials and settings but throttles
    itself to one request per second (the host sends no rate-limit headers).

    V3 is the default surface. Queries carrying ``linkProvider`` are routed to
    V2 because V3 silently ignores ``linkProvider``. Queries carrying
    ``from_date`` or ``to_date`` raise :class:`ConfigurationError` because V3
    returns zero rows for date filters and V2 silently ignores date parameters.
    """

    def __init__(self, client: BaseApiClient) -> None:
        self._v3 = ScholixV3Links(client)
        self._v2 = ScholixV2Links(client)
        self._client = client

    @staticmethod
    def _has_date_filter(filters: ScholixFilters | Mapping[str, Any] | None) -> bool:
        if filters is None:
            return False
        if isinstance(filters, ScholixFilters):
            return filters.from_date is not None or filters.to_date is not None
        return any(
            filters.get(key) is not None
            for key in ("from", "to", "from_date", "to_date")
        )

    @classmethod
    def _reject_date_filter(
        cls, filters: ScholixFilters | Mapping[str, Any] | None
    ) -> None:
        if cls._has_date_filter(filters):
            raise ConfigurationError(
                "Scholexplorer does not support 'from_date'/'to_date' filtering "
                "(V3 returns zero rows for 'from'/'to' and V2 silently ignores "
                "date parameters). Remove 'from_date'/'to_date' from filters and "
                "filter results by 'link_publication_date' client-side, or query "
                "Graph V4 research products with 'from_publication_date' / "
                "'to_publication_date'."
            )

    @staticmethod
    def _has_link_provider(filters: ScholixFilters | Mapping[str, Any] | None) -> bool:
        if filters is None:
            return False
        if isinstance(filters, ScholixFilters):
            return filters.linkProvider is not None
        return filters.get("linkProvider") is not None

        self, filters: ScholixFilters | Mapping[str, Any] | None

    """Client for the OpenAIRE Graph API (V4) and the Scholexplorer API.

    Entity resources are available as typed attributes; single-entity lookups
    go through the V3 by-id routes because V4 has none. Authentication
    resolves in the constructor: set ``openaire_refresh_token`` /
    ``refresh_token`` for OAuth2 refresh-token access, or set both
    ``openaire_client_id`` and ``openaire_client_secret`` (settings,
    ``AIRELOOM_OPENAIRE_CLIENT_ID`` / ``AIRELOOM_OPENAIRE_CLIENT_SECRET``, or
    ``client_id`` / ``client_secret``) for OAuth2 client-credentials access
    against the Keycloak token endpoint; partial client credentials raise
    :class:`ConfigurationError`, and without credentials the client is
    anonymous. An explicit ``auth_strategy`` always wins.

    The Scholix resource talks to the Scholexplorer host through a second
    internal client that shares settings and credentials but throttles itself
    to one request per second. See the package README and docs for usage
    examples of both the async and the sync variants.
    """

    def __init__(
        self,
        settings: AireloomSettings | None = None,
        auth: AuthStrategy | None = None,
        *,
        auth_strategy: AuthStrategy | None = None,
        base_url: str | None = None,
        scholix_base_url: str | None = None,
        mailto: str | None = None,
        client_id: str | None = None,
        client_secret: str | None = None,
        refresh_token: str | None = None,
        http_client: httpx.Client | None = None,
    ) -> None:
        """Build the client from settings, resolving credentials immediately.

        ``settings`` defaults to :func:`aireloom.config.get_settings`, which
        reads ``AIRELOOM_*`` environment variables and ``.env``. Keyword
        overrides (``base_url``, ``scholix_base_url``, ``mailto``,
        ``client_id``, ``client_secret``, ``refresh_token``) take precedence
        over ``settings``. Pass ``auth_strategy`` for a custom auth strategy
        (``auth`` is a deprecated alias).
        """
        if auth is not None and auth_strategy is not None:
            raise ConfigurationError(
                "Pass either 'auth_strategy' or the deprecated 'auth' parameter, "
                "not both."
            )
        if auth is not None:
            warnings.warn(
                "The 'auth' parameter is deprecated; use 'auth_strategy' instead.",
                DeprecationWarning,
                stacklevel=2,
            )
            explicit_auth: AuthStrategy | None = auth
        else:
            explicit_auth = auth_strategy

        resolved = settings or get_settings()
        overrides: dict[str, str] = {}
        if base_url is not None:
            overrides["base_url"] = base_url
        if scholix_base_url is not None:
            overrides["scholix_base_url"] = scholix_base_url
        if mailto is not None:
            overrides["mailto"] = mailto
        if client_id is not None:
            overrides["openaire_client_id"] = client_id
        if client_secret is not None:
            overrides["openaire_client_secret"] = client_secret
        if refresh_token is not None:
            overrides["openaire_refresh_token"] = refresh_token
        if overrides:
            resolved = resolved.model_copy(update=overrides)
        strategy = (
            explicit_auth if explicit_auth is not None else _resolve_auth(resolved)
        )
        super().__init__(
            base_url=resolved.base_url,
            settings=resolved,
            unwrapper=OpenAireUnwrapper(),
            auth=strategy,
            http_client=http_client,
        )
        self._scholix_api = SyncBaseApiClient(
            base_url=resolved.scholix_base_url,
            settings=resolved.model_copy(
                update={
                    "max_requests_per_second": resolved.scholix_max_requests_per_second
                }
            ),
            unwrapper=ScholixUnwrapper(),
            auth=strategy,
        )
        auth_label = (
            "anonymous" if isinstance(strategy, NoAuth) else type(strategy).__name__
        )
        logger.debug("%s initialized (%s).", type(self).__name__, auth_label)

    class Person(BaseEntity):
    """Model representing an OpenAIRE Person entity.

    Captures names, identifiers (``pid`` items are ``{value, typeCode,
    typeLabel}``, e.g. ORCID), biography, topics, measures, context, and
    co-authors. Inherits the ``id`` field from ``BaseEntity``.

    Reference: https://graph.openaire.eu/docs/api/entities/person
    """

    originalId: SafeList[str] = Field(default_factory=list)
    pid: SafeList[TypedPid] = Field(default_factory=list)
    givenName: SafeStr = ""
    familyName: SafeStr = ""
    alternativeNames: SafeList[str] = Field(default_factory=list)
    biography: SafeStr = ""
    subject: SafeList[PersonTopic] = Field(default_factory=list)
    indicator: SafeList[Measure] = Field(default_factory=list)

    consent: bool | None = None
    coAuthors: SafeList[str] = Field(default_factory=list)

    @computed_field
    @property
    def orcid(self) -> str | None:
        return extract_orcid(self.originalId, self.id, self.pid)

    @computed_field
    @property
    def full_name(self) -> str:
        parts = [n for n in (self.givenName, self.familyName) if n]
        return " ".join(parts)

    def __str__(self) -> str:
        name = self.full_name
        if name and self.orcid:
            return f"{name} (ORCID:{self.orcid})"
        if name:
            return name
        return f"Person(id={self.id!r})"
