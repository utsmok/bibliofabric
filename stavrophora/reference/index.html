    class StavrophoraClient(BaseApiClient):
    """Client for the Crossref API.

    ``StavrophoraClient`` drives async code and its generated twin
    ``StavrophoraSyncClient`` drives synchronous code; both expose the same
    resources and methods.

    Credentials are resolved in the constructor: an explicit ``auth_strategy``
    wins, then a Metadata Plus key (``crossref-api-key`` header), then
    ``mailto`` (polite pool via query parameter). When ``rate_limits``,
    ``max_concurrency``, and ``max_requests_per_second`` are not explicitly
    set on ``settings``, pool-aware defaults apply: Metadata Plus
    (``50/50`` req/s, concurrency ``10``), Polite Pool (``10/3`` req/s,
    concurrency ``3``), or Public Pool (``5/1`` req/s, concurrency ``1``).
    """

    def __init__(
        self,
        settings: StavrophoraSettings | None = None,
        *,
        mailto: str | None = None,
        plus_token: str | None = None,
        base_url: str | None = None,
        auth_strategy: AuthStrategy | None = None,
    ) -> None:
        resolved = settings or get_settings()
        mailto = mailto or resolved.mailto
        plus_token = plus_token or resolved.plus_token

        if plus_token:
            pool_rate_limits = _PLUS_RATE_LIMITS
            pool_concurrency = _PLUS_CONCURRENCY
        elif mailto:
            pool_rate_limits = _POLITE_RATE_LIMITS
            pool_concurrency = _POLITE_CONCURRENCY
        else:
            pool_rate_limits = _PUBLIC_RATE_LIMITS
            pool_concurrency = _PUBLIC_CONCURRENCY

        fields_set = resolved.model_fields_set
        updates: dict[str, Any] = {}
        if mailto and resolved.mailto != mailto:
            updates["mailto"] = mailto
        if (
            "rate_limits" not in fields_set
            and "max_requests_per_second" not in fields_set
        ):
            updates["rate_limits"] = dict(pool_rate_limits)
        if "max_concurrency" not in fields_set:
            updates["max_concurrency"] = pool_concurrency
        if updates:
            resolved = resolved.model_copy(update=updates)

        if auth_strategy is not None:
            auth = auth_strategy
        elif plus_token:
            auth = HeaderAuth(PLUS_TOKEN_HEADER, f"Bearer {plus_token}")
        elif mailto:
            auth = QueryParameterAuth("mailto", mailto)
        else:
            auth = NoAuth()
        super().__init__(
            base_url=base_url or CROSSREF_API_BASE_URL,
            settings=resolved,
            unwrapper=CrossrefUnwrapper(),
            auth=auth,
        )
        self._last_pool: str | None = None
        self._last_rate_limit_headers: dict[str, str] = {}

    @property
    def last_pool(self) -> str | None:
        """The ``x-api-pool`` header value from the most recent response, if any."""
        return self._last_pool

    @property
    def last_rate_limit_headers(self) -> dict[str, str]:
        """Crossref rate-limit headers captured from the most recent response."""
        return dict(self._last_rate_limit_headers)

    def on_response(self, response: httpx.Response) -> None:
        """Capture Crossref pool and rate-limit headers from each response."""
        captured: dict[str, str] = {}
        for header in _RATE_LIMIT_HEADERS:
            value = response.headers.get(header)
            if value is not None:
                captured[header] = value
        self._last_rate_limit_headers = captured
        self._last_pool = captured.get("x-api-pool")

    @cached_property
    def works(self) -> WorksResource:
        """The ``/works`` resource (search, iterate, batch DOI lookup, agency)."""
        return WorksResource(self)

    @cached_property
    def journals(self) -> JournalsResource:
        """The ``/journals`` resource."""
        return JournalsResource(self)

    @cached_property
    def funders(self) -> FundersResource:
        """The ``/funders`` resource."""
        return FundersResource(self)

    @cached_property
    def members(self) -> MembersResource:
        """The ``/members`` resource."""
        return MembersResource(self)

    @cached_property
    def types(self) -> TypesResource:
        """The ``/types`` resource."""
        return TypesResource(self)

    @cached_property
    def prefixes(self) -> PrefixesResource:
        """The ``/prefixes`` resource."""
        return PrefixesResource(self)

    @cached_property
    def licenses(self) -> LicensesResource:
        """The ``/licenses`` resource (list/iterate only)."""
        return LicensesResource(self)

    @cached_property
    def data_citations(self) -> DataCitationsResource:
        """The beta ``/beta/datacitations`` route (search, iterate, count)."""
        return DataCitationsResource(self)

    #: Only the top-level ``/works`` route is array-class. Everything else —
    #: single records, scoped ``/.../works`` routes, and all other routes —
    #: reports the ``polite-single`` pool in Crossref's ``x-api-pool`` header
    #: (2026-07-21 rate split; see the README and the phase 2 research). The
    #: beta data citations route reports its own ``datacitations`` pool.
    _TOP_LEVEL_WORKS = re.compile(r"^(/v1)?/works/?$")

    def rate_class(self, request: httpx.Request) -> str:
        """Map the request onto Crossref's polite pools (D-19)."""
        path = request.url.path
        if path.rstrip("/") == DATACITATIONS_PATH:
            return "datacitations"
        if self._TOP_LEVEL_WORKS.search(path):
            return "list"
        return "single"

        the works that updated them, each with a ``type`` and a timestamp or

            candidates = (
                [entry for entry in work.updated_by if entry.type in allowed]
                if allowed is not None
                else work.updated_by
            )
            if not candidates:
                break
            newest = max(candidates, key=_update_timestamp)
            target = newest.doi
            if not target or self.normalize_id(target) in seen:
                break
            seen.add(self.normalize_id(target))
            work = await self.get(target)
            if work.doi:
                seen.add(self.normalize_id(work.doi))
        return work

    class DataCitationsResource:
    """Typed client for the Crossref Data Citations API (beta).

    Every filter is optional and the filters combine with AND:
    ``subject_id`` (DOI of the citing work), ``object_id`` (DOI of the cited
    dataset or software), ``member_id`` (Crossref member of the citing
    work), ``relation`` (for example ``references``), and
    ``from_created_date`` / ``until_created_date`` (when Crossref added the
    relationship; ``2025``, ``2025-01``, ``2025-01-01``, or
    ``2025-01-01T12:00``).
    """

    def __init__(self, client: BaseApiClient) -> None:
        self._client = client
        self._url = str(httpx.URL(client.base_url).copy_with(path=DATACITATIONS_PATH))

    async def search(
        self,
        *,
        subject_id: str | None = None,
        object_id: str | None = None,
        member_id: str | int | None = None,
        relation: str | None = None,
        from_created_date: str | None = None,
        until_created_date: str | None = None,
        rows: int | None = None,
        page: int | None = None,
    ) -> Page[DataCitation]:
        """Return one page of data citations.

        ``page`` is 0-based. ``page.next_cursor`` holds the next page number
        as a string, or ``None`` after the last page; ``page.total`` is the
        number of matching relationships.
        """
        params = _params(
            subject_id=subject_id,
            object_id=object_id,
            member_id=member_id,
            relation=relation,
            from_created_date=from_created_date,
            until_created_date=until_created_date,
            rows=rows,
            page=page,
        )
        message, response = await self._fetch(params)
        results = [
            validate_model(
                DataCitation, item, request=response.request, response=response
            )
            for item in message.get("items") or []
        ]
        next_page = message.get("next-page")
        total = message.get("total-results")
        meta = {key: value for key, value in message.items() if key != "items"}
        return Page(
            results=results,
            total=int(total) if total is not None else None,
            next_cursor=str(next_page) if next_page is not None else None,
            page=page if page is not None else 0,
            meta=meta,
        )

    async def iterate(
        self,
        *,
        subject_id: str | None = None,
        object_id: str | None = None,
        member_id: str | int | None = None,
        relation: str | None = None,
        from_created_date: str | None = None,
        until_created_date: str | None = None,

                    DataCitation, item, request=response.request, response=response

                "Crossref /beta/datacitations answered without a message object."

        Older records of an update chain carry ``updated-by`` entries naming
        the works that updated them, each with a ``type`` and a timestamp or
        date parts. By default only ``("new_version", "new_edition")`` entries

                [entry for entry in work.updated_by if entry.type in allowed]
                if allowed is not None
                else work.updated_by
            )
            if not candidates:
                break

            target = newest.doi
            if not target or self.normalize_id(target) in seen:
                break
            seen.add(self.normalize_id(target))
            work = self.get(target)
            if work.doi:
                seen.add(self.normalize_id(work.doi))
        return work

    class SyncDataCitationsResource:
    """Typed client for the Crossref Data Citations API (beta).

    Every filter is optional and the filters combine with AND:
    ``subject_id`` (DOI of the citing work), ``object_id`` (DOI of the cited
    dataset or software), ``member_id`` (Crossref member of the citing
    work), ``relation`` (for example ``references``), and
    ``from_created_date`` / ``until_created_date`` (when Crossref added the
    relationship; ``2025``, ``2025-01``, ``2025-01-01``, or
    ``2025-01-01T12:00``).
    """

    def __init__(self, client: SyncBaseApiClient) -> None:
        self._client = client
        self._url = str(httpx.URL(client.base_url).copy_with(path=DATACITATIONS_PATH))

    def search(
        self,
        *,
        subject_id: str | None = None,
        object_id: str | None = None,
        member_id: str | int | None = None,
        relation: str | None = None,
        from_created_date: str | None = None,
        until_created_date: str | None = None,
        rows: int | None = None,
        page: int | None = None,
    ) -> Page[DataCitation]:
        """Return one page of data citations.

        ``page`` is 0-based. ``page.next_cursor`` holds the next page number
        as a string, or ``None`` after the last page; ``page.total`` is the
        number of matching relationships.
        """
        params = _params(
            subject_id=subject_id,
            object_id=object_id,
            member_id=member_id,
            relation=relation,
            from_created_date=from_created_date,
            until_created_date=until_created_date,
            rows=rows,
            page=page,
        )
        message, response = self._fetch(params)
        results = [
            validate_model(
                DataCitation, item, request=response.request, response=response
            )
            for item in message.get("items") or []
        ]
        next_page = message.get("next-page")
        total = message.get("total-results")
        meta = {key: value for key, value in message.items() if key != "items"}
        return Page(
            results=results,
            total=int(total) if total is not None else None,
            next_cursor=str(next_page) if next_page is not None else None,
            page=page if page is not None else 0,
            meta=meta,
        )

    def iterate(
        self,
        *,
        subject_id: str | None = None,
        object_id: str | None = None,
        member_id: str | int | None = None,
        relation: str | None = None,
        from_created_date: str | None = None,
        until_created_date: str | None = None,
        rows: int = DATACITATIONS_MAX_ROWS,
    ) -> Iterator[DataCitation]:
        """Yield every matching data citation, following ``next-page``."""
        page_number: Any = 0
        while page_number is not None:
            params = _params(
                subject_id=subject_id,
                object_id=object_id,
                member_id=member_id,
                relation=relation,
                from_created_date=from_created_date,
                until_created_date=until_created_date,
                rows=rows,
                page=int(page_number),
            )
            message, response = self._fetch(params)
            items = message.get("items") or []
            for item in items:
                yield validate_model(
                    DataCitation, item, request=response.request, response=response
                )
            page_number = message.get("next-page") if items else None

    def _fetch(self, params: dict[str, Any]) -> tuple[dict[str, Any], httpx.Response]:
        """Send one request and return the ``message`` object and the response."""
        response = self._client.request("GET", self._url, params=params)
        body = self._client.parse_json(response)
        message = body.get("message") if isinstance(body, dict) else None
        if not isinstance(message, dict):
            raise ConfigurationError(
                "Crossref /beta/datacitations answered without a message object."
            )
        return message, response

    def retractions(self) -> list[UpdateTo]:
        """The ``updated-by`` entries of type ``retraction``.
