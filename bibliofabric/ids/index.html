    Identifiers

    bibliofabric.ids holds the canonical identifier normalizers shared by all clients. They never URL-encode; clients add API-specific encoding on top.

    Function Behavior
    normalize_doi Strips https://doi.org/, dx.doi.org, and doi: prefixes, trims, lowercases
    is_doi Whether the value looks like a bare DOI
    normalize_orcid Strips prefixes, returns the hyphenated 16-character form
    normalize_ror Strips prefixes, lowercases
    normalize_issn Returns the hyphenated uppercase form (2049-3630)
    normalize_isbn A valid ISBN-10 converts to its 978-prefixed ISBN-13 with a recomputed check digit; anything else comes back with hyphens and spaces removed and uppercased
    normalize_isni Four groups of four characters (0000 0001 2109 7254)
    normalize_arxiv Bare id; old-style ids lowercased; version suffix kept
    normalize_pmid Bare digits; strips https://pubmed.ncbi.nlm.nih.gov/, pubmed:, pmid:
    normalize_pmcid PMC1234567; bare digits get the PMC prefix
    normalize_wikidata_qid Q42 or a property id such as P3153; strips wikidata:, wd:, wikidata.org URLs, and a #fragment
    normalize_gnd Bare id; a trailing x check digit becomes X
    normalize_viaf Bare cluster number without leading zeros
    strip_url_prefix Removes the first matching prefix, case-insensitively

    API reference

    bibliofabric.ids

    Canonical identifier normalization shared by bibliofabric-family clients.

    Clients add API-specific encoding on top of these normalizers; the normalizers themselves never URL-encode. Each one strips the resolver URLs and scheme prefixes its identifier is written with and returns the canonical form. A value that does not match the known shape comes back stripped and tidied, so unvetted API fields can pass through without raising.

    def normalize_arxiv(value: str) -> str:
    """Return the bare arXiv id (``1706.03762`` or ``hep-th/9901001``).

    Version suffixes are kept as given; old-style ids are lowercased.
    """
    stripped = strip_url_prefix(value, _ARXIV_PREFIXES).rstrip("/")
    category, slash, tail = stripped.partition("/")
    return f"{category.lower()}/{tail.lower()}" if slash else stripped

    def normalize_isbn(value: str) -> str:
    """Return the ISBN in ISBN-13 form without hyphens (``9780306406157``).

    A valid ISBN-10 converts to its ``978``-prefixed ISBN-13 with a recomputed
    check digit. Any other value comes back compacted: hyphens and spaces
    removed, uppercased.
    """
    compact = _compact(strip_url_prefix(value, _ISBN_PREFIXES))
    if not _is_valid_isbn_10(compact):
        return compact
    stem = f"978{compact[:-1]}"
    return f"{stem}{_isbn_13_check_digit(stem)}"

    def normalize_isni(value: str) -> str:
    """Return the ISNI as four groups of four characters (``0000 0001 2109 7254``)."""
    compact = _compact(strip_url_prefix(value, _ISNI_PREFIXES))
    if (
        len(compact) == _ISNI_LENGTH
        and compact[:-1].isdigit()
        and compact[-1] in _DIGITS_X
    ):
        return " ".join(
            compact[index : index + 4] for index in range(0, _ISNI_LENGTH, 4)
        )
    return compact

    def normalize_issn(value: str) -> str:
    """Return the ISSN in hyphenated uppercase form (``2049-3630``)."""
    stripped = value.strip().upper()
    compact = _compact(stripped)
    if (
        len(compact) == _ISSN_LENGTH
        and compact[:4].isdigit()
        and compact[4:-1].isdigit()
        and compact[-1] in "0123456789X"
    ):
        return f"{compact[:4]}-{compact[4:]}"
    return stripped

    def strip_url_prefix(value: str, prefixes: Iterable[str]) -> str:
    """Remove the first matching prefix (case-insensitive) and trim whitespace."""
    stripped = value.strip()
    for prefix in prefixes:
        if stripped.lower().startswith(prefix.lower()):
            return stripped[len(prefix) :].strip()
    return stripped
